Google Sheets connection
One-time setup: Follow these steps to connect this app to your Google Sheet.

1. Open your Google Sheet → Extensions → Apps Script
2. Replace all code with the script below → click Save
3. Click Deploy → New deployment → Web app
4. Set "Who has access" to Anyone → click Deploy
5. Copy the Web App URL and paste it below
function doPost(e) { var sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName("Receipts"); if (!sheet) { sheet = SpreadsheetApp .getActiveSpreadsheet() .insertSheet("Receipts"); sheet.appendRow([ "Receipt No","Date","Month Paid For", "Student Name","Tuition Fee (RM)", "Book Fee (RM)","Total (RM)", "Payment Method","Center Name" ]); } var d = JSON.parse(e.postData.contents); sheet.appendRow([ d.num, d.date, d.monthLabel, d.student, d.tuition, d.book, d.total, d.method, d.center ]); return ContentService .createTextOutput(JSON.stringify({ok:true})) .setMimeType(ContentService.MimeType.JSON); }
